cmake_minimum_required(VERSION 3.8)
project(detection_fusion)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(wuta_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(ament_cmake_python REQUIRED)
add_executable(detection_fusion_node_cpp src/detection_fusion_node_cpp.cpp)
target_link_libraries(detection_fusion_node_cpp Eigen3::Eigen)
ament_target_dependencies(detection_fusion_node_cpp rclcpp wuta_msgs sensor_msgs std_msgs visualization_msgs tf2_ros)
install(TARGETS detection_fusion_node_cpp DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/detection_fusion_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(fusion_core test/test_core.py)
  ament_add_pytest_test(calibration_contract test/test_calibration.py)
endif()
ament_package()
